**Tech Stack:** React 19, TanStack Start/Router, TypeScript, CSS custom properties, Bun test, Playwright, local Supabase QA fixture

Create `tests/e2e/member-account-premium-layout.spec.mjs`:

```js
/**
 * Local-only authenticated visual/layout regression for the premium member account page.
 * It uses the real /auth flow and local fixture, and never submits a deletion request.
 */
import { existsSync } from "node:fs";
import { mkdir, readFile } from "node:fs/promises";
import { resolve } from "node:path";
import { chromium } from "playwright";

const baseUrl = process.env.APP_BASE_URL || "http://127.0.0.1:4176";
const fixtureEmail = "qa-member-ui@cloudcore.test";
const localChrome = "/Applications/Google Chrome.app/Contents/MacOS/Google Chrome";
const screenshotDir = resolve(
  process.cwd(),
  "docs/ui-ux-audit/screenshots/after/member-account-premium-polish",
);
const cases = [
  { lang: "en", dir: "ltr", width: 320, height: 720 },
  { lang: "en", dir: "ltr", width: 390, height: 844 },
  { lang: "en", dir: "ltr", width: 1440, height: 900 },
  { lang: "he", dir: "rtl", width: 390, height: 844 },
  { lang: "he", dir: "rtl", width: 1440, height: 900 },
  { lang: "ar", dir: "rtl", width: 390, height: 844 },
  { lang: "ar", dir: "rtl", width: 1440, height: 900 },
];

function assertLoopbackTarget(value) {
  const host = new URL(value).hostname.replace(/^\[|\]$/g, "");
  if (!new Set(["127.0.0.1", "localhost", "::1"]).has(host)) {
    throw new Error("member_account_premium_qa_refuses_non_local_target");
  }
}

function parseEnv(contents) {
  return Object.fromEntries(
    contents
      .split(/\r?\n/)
      .filter((line) => line && !line.startsWith("#") && line.includes("="))
      .map((line) => {
        const index = line.indexOf("=");
        const value = line.slice(index + 1).replace(/^(\"|')(.*)\1$/, "$2");
        return [line.slice(0, index), value];
      }),
  );
}

async function signIn(page, password) {
  await page.goto(`${baseUrl}/auth`, { waitUntil: "networkidle" });
  await page.getByRole("button", { name: "English" }).click();
  await page.getByLabel(/email/i).fill(fixtureEmail);
  await page.getByRole("textbox", { name: /password/i }).fill(password);
  await page.locator('form button[type="submit"]').click();
  await page.waitForURL(/\/member/, { timeout: 10_000 });
}

async function assertAccountLayout(page, expected) {
  await page.goto(`${baseUrl}/member/account`, { waitUntil: "networkidle" });
  await page.locator("#profile-language").selectOption(expected.lang);
  await page.waitForFunction(
    ({ lang, dir }) =>
      document.documentElement.lang === lang && document.documentElement.dir === dir,
    expected,
  );
  await page.getByRole("heading", { level: 1 }).waitFor({ state: "visible" });
  if (!(await page.locator("main").innerText()).includes("Member QA")) {
    throw new Error(`member_account_fixture_not_rendered:${expected.lang}`);
  }

  const geometry = await page.evaluate(() => {
    const content = document.querySelector(".member-account-page__content");
    const links = [...document.querySelectorAll(".member-account-legal-links a")];
    const panel = document.querySelector(".member-danger-zone");
    const action = document.querySelector("#account-delete-request");
    if (!content || links.length !== 3 || !panel || !action) {
      throw new Error("member_account_premium_layout_missing_required_element");
    }
    const contentRect = content.getBoundingClientRect();
    const panelRect = panel.getBoundingClientRect();
    const actionRect = action.getBoundingClientRect();
    const linkRects = links.map((link) => link.getBoundingClientRect());
    return {
      noOverflow: document.documentElement.scrollWidth <= document.documentElement.clientWidth,
      contentWidth: contentRect.width,
      panelInside:
        panelRect.left >= -0.5 && panelRect.right <= document.documentElement.clientWidth + 0.5,
      actionInside:
        actionRect.left >= panelRect.left - 0.5 && actionRect.right <= panelRect.right + 0.5,
      actionHeight: actionRect.height,
      linkHeights: linkRects.map((rect) => rect.height),
      legalGap: getComputedStyle(document.querySelector(".member-account-legal-links")).gap,
      panelDisplay: getComputedStyle(panel).display,
    };
  });

  if (!geometry.noOverflow) throw new Error(`horizontal_overflow:${expected.lang}:${expected.width}`);
  if (!geometry.panelInside || !geometry.actionInside) {
    throw new Error(`deletion_panel_overflow:${expected.lang}:${expected.width}`);
  }
  if (geometry.panelDisplay !== "grid") throw new Error("deletion_panel_is_not_grid");
  if (geometry.legalGap === "0px") throw new Error("legal_actions_touch");
  if (geometry.linkHeights.some((height) => height < 44) || geometry.actionHeight < 44) {
    throw new Error(`account_touch_target_too_small:${expected.lang}:${expected.width}`);
  }
  if (expected.width >= 1000 && geometry.contentWidth > 961) {
    throw new Error(`desktop_column_too_wide:${geometry.contentWidth}`);
  }

  const trigger = page.locator("#account-delete-request");
  await trigger.focus();
  await trigger.press("Enter");
  const dialog = page.getByRole("alertdialog");
  await dialog.waitFor({ state: "visible" });
  const dialogBox = await dialog.boundingBox();
  if (!dialogBox || dialogBox.x < 0 || dialogBox.x + dialogBox.width > expected.width + 1) {
    throw new Error(`deletion_dialog_overflow:${expected.lang}:${expected.width}`);
  }
  await page.keyboard.press("Escape");
  if (!(await trigger.evaluate((element) => element === document.activeElement))) {
    throw new Error(`deletion_focus_not_restored:${expected.lang}:${expected.width}`);
  }

  await page.screenshot({
    path: resolve(screenshotDir, `account-premium-${expected.lang}-${expected.width}.png`),
    fullPage: true,
  });
}

async function main() {
  assertLoopbackTarget(baseUrl);
  const variables = parseEnv(await readFile(resolve(process.cwd(), ".env.qa.local"), "utf8"));
  if (!variables.QA_MEMBER_UI_UX_PASSWORD) {
    throw new Error("member_account_premium_fixture_password_missing");
  }
  await mkdir(screenshotDir, { recursive: true });
  const browser = await chromium.launch({
    headless: true,
    ...(existsSync(localChrome) ? { executablePath: localChrome } : {}),
  });
  try {
    const context = await browser.newContext({ viewport: { width: 390, height: 844 } });
    const page = await context.newPage();
    await signIn(page, variables.QA_MEMBER_UI_UX_PASSWORD);
    for (const expected of cases) {
      await page.setViewportSize({ width: expected.width, height: expected.height });
      await assertAccountLayout(page, expected);
    }
    console.log(`member-account-premium-layout-pass:${cases.length}`);
  } finally {
    await browser.close();
  }
}

main().catch((error) => {
  console.error(
    `member-account-premium-layout-failed:${error instanceof Error ? error.message : "unknown"}`,
  );
  process.exit(1);
});
```

The test changes language through the real profile control, opens and closes the real dialog, and never activates confirmation.
